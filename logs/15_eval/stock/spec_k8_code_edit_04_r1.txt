def process(items):
    """
    Processes a list of items by applying specific transformations and calculating a total.

    Parameters:
        items (list): A list of elements. Each element can be an integer or None.

    Returns:
        tuple: A tuple containing two elements:
            - list: A list of transformed values based on the input items.
            - int: The sum of all transformed values.

    Description:
        - For each item in the input list:
            - If the item is not None and greater than 0, multiply it by 2.
            - Otherwise, append 0 to the result list.
        - Sum all values in the result list and return the result list and the total.

    Example:
        >>> process([1, None, 3, -2, 5])
        ([2, 0, 6, 0, 10], 18)
    """
    # Input validation
    if not isinstance(items, list):
        raise TypeError("Input must be a list.")
    if not all(isinstance(item, (int, type(None))) for item in items):
        raise TypeError("All items in the list must be integers or None.")

    result = []
    for i in range(len(items)):
        if items[i] is not None:
            if items[i] > 0:
                result.append(items[i] * 2)
            else:
                result.append(0)
    total = sum(result)
    return result, total
