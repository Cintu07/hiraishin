```python
def process(items):
    """
    Processes a list of items by applying specific transformations and calculating a total.

    Parameters:
    -----------
    items : list
        A list of elements. Each element can be an integer or None.

    Returns:
    --------
    tuple
        A tuple containing two elements:
        - list: The transformed list where:
            - If the item is not None and greater than 0, it is doubled.
            - Otherwise, it is replaced with 0.
        - int: The sum of all elements in the transformed list.

    Raises:
    -------
    TypeError
        If the input is not a list.
    ValueError
        If any element in the list is not an integer or None.
    """
    # Input validation
    if not isinstance(items, list):
        raise TypeError("Input must be a list.")

    for item in items:
        if not isinstance(item, (int, type(None))):
            raise ValueError("All elements must be integers or None.")

    result = []
    for i in range(len(items)):
        if items[i] is not None:
            if items[i] > 0:
                result.append(items[i] * 2)
            else:
                result.append(0)
        else:
            result.append(0)

    total = sum(result)
    return result, total
```
